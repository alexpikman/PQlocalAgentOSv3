  function readWorkspace() {
    try {
      const saved = JSON.parse(localStorage.getItem(STORAGE_KEY) || '{}');
      state.projects = Array.isArray(saved.projects) ? saved.projects : [];
      state.conversations = Array.isArray(saved.conversations) ? saved.conversations : [];
      state.activeConversationId = saved.activeConversationId || '';
      state.activeProjectId = saved.activeProjectId || '';
    } catch (_) {
      state.projects = [];
      state.conversations = [];
    }
  }

  function saveWorkspace() {
    try {

        activeConversationId: state.activeConversationId,
        activeProjectId: state.activeProjectId

  function handleChatEvent(event, replay = false) {
    const payload = event.payload || {};
    if (state.renderer) state.renderer.apply(event);
    if (event.type === 'user_message') {
      const commandId = payload.command_id || '';
      if (!state.messages.some(message => commandId && message.commandId === commandId)) {
        state.messages.push({ role: 'user', content: String(payload.text || ''), commandId, eventId: event.event_id, time: formatTime(new Date(event.created_at || Date.now())) });
      }
    } else if (event.type === 'assistant_delta') {
      let message = state.messages.find(item => item.messageId === event.message_id);
      if (!message) {
        message = { role: 'assistant', content: '', messageId: event.message_id, time: formatTime(new Date(event.created_at || Date.now())) };
        state.messages.push(message);
      }
      message.content += String(payload.text || '');
    } else if (event.type === 'assistant_message') {
      let message = state.messages.find(item => item.messageId === event.message_id);
      if (!message) {
        message = { role: 'assistant', content: '', messageId: event.message_id, time: formatTime(new Date(event.created_at || Date.now())) };
        state.messages.push(message);
      }
      message.content = String(payload.content || '');
      if (message.content.trimStart().startsWith('{')) {
        try {
          const parsed = JSON.parse(message.content.trim());
          if (parsed && typeof parsed === 'object' && !Array.isArray(parsed)) {
            const type = String(parsed.type || '').toLowerCase();
            if (type === 'message' || type === 'assistant_message') message.content = String(parsed.text ?? parsed.content ?? '');
            else if (type === 'tool_call' || type === 'tool-call' || (typeof parsed.name === 'string' && ('parameters' in parsed || 'arguments' in parsed))) {
              message.content = '';
              message.tool_calls = [{ name: parsed.name || 'РРЅСЃС‚СЂСѓРјРµРЅС‚', arguments: parsed.parameters ?? parsed.arguments ?? {}, id: '' }];
            } else if (typeof parsed.text === 'string') message.content = parsed.text;
            else message.content = 'РџРѕР»СѓС‡РµРЅ СЃС‚СЂСѓРєС‚СѓСЂРёСЂРѕРІР°РЅРЅС‹Р№ РѕС‚РІРµС‚ Р±РµР· СЂР°СЃРїРѕР·РЅР°РЅРЅРѕРіРѕ С‚РµРєСЃС‚Р°.';
          }
        } catch (_) {}
      }
      if (Array.isArray(payload.tool_calls)) message.tool_calls = payload.tool_calls.map(call => ({ name: call?.function?.name || call?.name || 'РРЅСЃС‚СЂСѓРјРµРЅС‚', arguments: call?.function?.arguments || call?.arguments || {}, id: call?.id || '' }));
    } else if (event.type === 'tool_call') {
      const message = [...state.messages].reverse().find(item => item.role === 'assistant');
      if (message) {
        message.tool_calls = Array.isArray(message.tool_calls) ? message.tool_calls : [];
        const toolName = payload.name || 'РРЅСЃС‚СЂСѓРјРµРЅС‚';
        const existingCall = message.tool_calls.find(item => item.id === event.tool_call_id || (!item.id && item.name === toolName));
        if (existingCall) {
          existingCall.id = event.tool_call_id;
          existingCall.arguments = payload.arguments || existingCall.arguments || {};
          existingCall.status = 'running';
        } else {
          message.tool_calls.push({ id: event.tool_call_id, name: toolName, arguments: payload.arguments || {}, status: 'running' });
        }
      }
    } else if (event.type === 'tool_result') {
      for (const message of state.messages) {
        for (const call of message.tool_calls || []) if (call.id === event.tool_call_id) call.status = (payload.status === 'error' || payload.status === 'denied' || payload.error) ? 'error' : 'completed';
      }
    } else if (event.type === 'error') {
      if (payload.code === 'verification_required') { setSending(false); elements.input?.focus(); }
      showNotice(payload.message || 'РћС€РёР±РєР° РІС‹РїРѕР»РЅРµРЅРёСЏ Р°РіРµРЅС‚Р°.', 'warning');
    } else if (event.type === 'turn_completed' || event.type === 'turn_cancelled') {
      setSending(false); if (!replay) saveEventConversation(); elements.input?.focus();
    }
    if (!replay && event.type !== 'assistant_delta') saveWorkspace();
  }

  async function setupEventTransport(conversation) {

        await chatApi('/chat/import', { method: 'POST', body: JSON.stringify({ conversation_id: conversation.serverConversationId, messages: legacyMessages.map(message => ({ role: message.role, content: String(message.content || '') })) }) });
      }
      saveWorkspace();
    }
    if (state.transport && state.transportConversationId === conversation.serverConversationId) return state.transport;
    state.transport?.close();

      import('/chat-transport.js'), import('/chat-renderer.js')
    ]);
    elements.messages.replaceChildren();
    state.messages = [];

        };
      });
    const activityEl = document.getElementById('chat-activity');
    if (activityEl) { activityEl.hidden = true; activityEl.dataset.state = ''; }
    const transport = new ChatTransport({
      conversationId: conversation.serverConversationId,
      onEvent: event => handleChatEvent(event, false),

      }

    for (const event of (history.events || [])) handleChatEvent(event, true);
    const lastSequence = (history.events || []).reduce((max, event) => Math.max(max, Number(event.sequence) || 0), 0);
    state.transport = transport; state.transportConversationId = conversation.serverConversationId; window.PQChatTransport = transport;

    if (activeConversation()?.id === conversation.id) {
      conversation.messages = state.messages; conversation.agentId = state.agent;
    }
    saveWorkspace();
    return transport;
  }

  async function sendMessage() {
    const content = elements.input.value.trim();
    if (!content || state.sending || !state.agent) return;
    clearNotice();
    setSending(true);
    elements.input.value = ''; resizeInput();
    let contentCommandId = '';
    try {
      const conversation = activeConversation();
      const transport = await setupEventTransport(conversation);
      const commandId = (window.crypto && crypto.randomUUID) ? crypto.randomUUID() : ('cmd-' + Date.now() + '-' + Math.random().toString(16).slice(2));
      contentCommandId = commandId;
      const mode = conversation?.mode || (conversation?.projectId ? 'project' : 'chat');
      if (mode === 'project' && !conversation?.projectId) {
        setSending(false);
        showNotice('Р’С‹Р±РµСЂРёС‚Рµ РїСЂРѕРµРєС‚ РґР»СЏ СЂРµР¶РёРјР° В«Р’РќРЈРўР Р РџР РћР•РљРўРђВ».', 'warning');
        elements.input.value = content; resizeInput();
        return;
      }
      state.renderer?.startTurn(Date.now());
      window.dispatchEvent(new CustomEvent('pq-chat-user-send-status', { detail: { commandId, status: 'pending' } }));
      await transport.sendMessage(content, commandId, state.agent, mode === 'project' ? conversation.projectId : '', mode);
      window.dispatchEvent(new CustomEvent('pq-chat-user-send-status', { detail: { commandId, status: 'success' } }));
    } catch (error) {
      setSending(false);
      if(contentCommandId){
        window.dispatchEvent(new CustomEvent('pq-chat-user-send-status', { detail: { commandId: contentCommandId, status: 'error' } }));
        state.renderer?.showLocalUserMessage(content, contentCommandId, 'error');
      }
      showNotice(error.message || 'РќРµ СѓРґР°Р»РѕСЃСЊ РѕС‚РїСЂР°РІРёС‚СЊ СЃРѕРѕР±С‰РµРЅРёРµ.', 'warning');
      elements.input.value = content; resizeInput();
    } finally {
      elements.input.focus();
    }
  }

      esc: value => String(value ?? '').replace(/[&<>"']/g, char => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[char])),
      toast: message => showNotice(message, 'info'),
      refresh: async () => {}
    });
  }

  window.ChatPage = { init };

  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', init, { once: true });
  } else {
    init();
  }
})();
